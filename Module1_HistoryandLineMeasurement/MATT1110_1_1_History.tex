\begin{frame}
    \frametitle{Editorial Note on the Pyramids}

    \begin{itemize}[<+- | alert@+>]
        \item We often hear ``we don't know how the pyramids were built''
        \item That doesn't mean the builders had advanced tech, were time travelers, had alien intervention
        \item There are literally hundreds of possible, readily available methods--we just don't know \emph{which one} they used
    \end{itemize}

    \begin{quote}
        I have dwelt longer upon the history of the Samians than I should otherwise have done, because they are responsible for three of the greatest building and engineering feats in the Greek world: the first is a tunnel nearly a mile long, eight feet wide and eight feet high, driven clean through the base of a hill nine hundred feet in height. \section{The Evolution of Linear Measurement}

\begin{frame}
	\frametitle{The Foot and the Inch}

    \begin{itemize}[<+- | alert@+>]
        \item Over a bit of time the cubit start to be called a \emph{foot}
        \item 1 foot was made up of 12 thumbnail widths (\emph{unicae})
        \item The unicae would eventually become the \emph{inch}
    \end{itemize}
\end{frame}

\begin{frame}
	\frametitle{The Timeline (1 of 2)}
    \begin{itemize}[<+- | alert@+>]
        \item[12$^{th}$ century] 1 yard = Distance from the tip of King Henry I's nose to the end of his thumb
        \item[1324] 1 inch = length of three barleycorns end to end
        \item[1588] Queen Elizabeth I commissioned a brass bar defined to be a standard yard
        \begin{itemize}
            \item Remained the standard until 1878
        \end{itemize}
    \end{itemize}
\end{frame}

\begin{frame}
	\frametitle{The Timeline (2 of 2)}
    \begin{itemize}[<+- | alert@+>]
        \item[16$^{th}$ century] The foot ranged from 9-3/4" to 19"
        \item[1535] James K\"{o}bel, German mathematician and surveyor:
        \begin{quote}
            Stand at the door of a church on a Sunday and bid 16 men to stop, tall ones and small ones, as they happen to pass out when the service is finished; then make them put their left feet one behind the other, and the length thus obtained shall be a right and lawful rood to measure and survey the land with, and the 16th part thereof shall be one right and lawful foot.
        \end{quote}
    \end{itemize}
\end{frame}

\section{Measurement Systems}

\begin{frame}
	\frametitle{The Metric System}
    \begin{itemize}[<+- | alert@+>]
        \item French scientists wanted permanent measurement system tied to something that didn't change
        \item[1793] Measured Earth's diameter, divided distance from equator to North Pole into 10 million parts = 1 metre
        \item[Current] 1 meter = The distance travelled by light in a vacuum in 1/299,792,458 second
        \item The core tenet of the metric system is that units \emph{must} be based on something that doesn't change
    \end{itemize}
\end{frame}

    \begin{itemize}[<+- | alert@+>]

        \item[1866] Congress legalized the metric system, 1 meter = 39.37 inches

    \end{itemize}

\section{Measurement Terms}

\begin{frame}
	\frametitle{Dimension}

    The measure of width, height, depth, or length in units of length of any object
\end{frame}

\begin{frame}
	\frametitle{Basic Dimension}
    
    The exact theoretical size from which all limiting variations are made 
\end{frame}

\begin{frame}
	\frametitle{Tolerance}
    
    The permissible variation in the size of a part
    
    \begin{columns}[t]
        \begin{column}{0.48\linewidth}
            \begin{block}{Bilateral}
                $1.000" \pm 0.002"$
                $1.00" +0.001"/-0.003"$
            \end{block}
        \end{column}
        \begin{column}{0.48\linewidth}
            \begin{block}{Unilateral}
                $1.000" +0.000"/-0.004"$
                $1.000" +0.004"/-0.000"$
            \end{block}
        \end{column}
    \end{columns}
    \vspace{1em}
    \centering
    All of these have a tolerance of 0.004"
\end{frame}

\begin{frame}
	\frametitle{Limits}

    The maximum and minimum dimensions obtained by applying the tolerances to the basic dimensions, the extreme dimensions beyond which the work cannot extend

    \begin{itemize}
        \item{2.251" $\pm$ 0.001"}
        \item{Lower limit = 2.251" - 0.001" = 2.250"}
        \item{Upper limit = 2.251" + 0.001" = 2.252"}
    \end{itemize}  
\end{frame}

\begin{frame}
	\frametitle{Mean Dimension}
    \begin{description}[<+- | alert@+>]
        \item[Mean Dimension]{average of the high and low limits}
            \begin{itemize}
                \item{0.250" +0.002"/-0.004"}
                \item{Lower limit = 0.250" - 0.004" = 0.246"}
                \item{Upper limit = 0.250" + 0.002" = 0.252"}
                \item{$\dfrac{0.252"+0.246"}{2}=0.249"$}
            \end{itemize}
    \end{description}    
\end{frame}

\begin{frame}
	\frametitle{Standard Size}

    A series of recognized or accepted sizes corresponding to various subdivisions of a recognized unit of length

    \begin{itemize}[<+- | alert@+>]
        \item Bar stock -- 1/64 inch, 1/32 inch, etc.
        \item Non-standard sizes are considered ``odd sizes''
        \item Note that standard sizes may not match their dimensions
        \begin{itemize}
            \item[Lumber] 2x4 = 1.5x3.5
            \item[Plumbing] 3" pipe = 3" inner diameter
            \item[Conduit] 1" conduit = 1" outer diameter  
        \end{itemize}
        \item Standard sizes can be arbitrary letters or numbers
            \begin{itemize}
                \item[Paper] A4 = 210mm x 297mm
            \end{itemize}
    \end{itemize}   
\end{frame}
